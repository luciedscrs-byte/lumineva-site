Notre réponse

En partie, dès maintenant. Les 693 abonnés sans commande peuvent être ciblés tout de suite : ils forment un segment Shopify, et un envoi unique leur rappelle le code. Pour les futurs inscrits, un Flow envoie le rappel tout seul. Ce qu’on ne peut pas faire : savoir d’où vient chaque abonné ni qui a déjà vu un code. Il faut des étiquettes pour ça.

def wilson(k, n_, z=1.96):
    if n_ == 0: return [None, None]
    p = k / n_; d = 1 + z * z / n_
    c = (p + z * z / (2 * n_)) / d; h = z * math.sqrt(p * (1 - p) / n_ + z * z / (4 * n_ * n_)) / d
    return [100 * (c - h), 100 * (c + h)]


def fisher(a, na, b, nb):

def hazard(statut, T, H=30):
    """Observations = (client, commande) où le client a déjà passé `statut` commandes. Silence de T jours depuis cette commande ;
    succès = nouvelle commande dans les H jours suivants. On garde les observations suivies au moins T+H jours."""
    ok = tot = 0
    for v in clients.values():
        for i in range(len(v)):

            if nxt is not None and (nxt - last).days <= T: continue
            tot += 1; ok += 1 if (nxt is not None and (nxt - last).days <= T + H) else 0
    return {'T': T, 'retours': ok, 'observations': tot, 'taux': 100 * ok / tot if tot else None, 'ic': wilson(ok, tot)}

def perm_diff(vals_a, vals_b, w_a=None, w_b=None, n_perm=20000, seed=3):
    """écart des taux de clic entre deux groupes, test de permutation exact/approché sur les étiquettes (pondéré par le nombre d'envois)"""
    allv = np.array(vals_a + vals_b, dtype=float); w = np.array((w_a or [1] * len(vals_a)) + (w_b or [1] * len(vals_b)), dtype=float)
    na = len(vals_a); obs = np.average(allv[:na], weights=w[:na]) - np.average(allv[na:], weights=w[na:])
    n = len(allv)
    if len(list(itertools.combinations(range(n), na))) <= 50000:
        diffs = []
        for comb in itertools.combinations(range(n), na):
            m = np.zeros(n, bool); m[list(comb)] = True
            diffs.append(np.average(allv[m], weights=w[m]) - np.average(allv[~m], weights=w[~m]))
        diffs = np.array(diffs)
    else:
        rng = np.random.default_rng(seed); diffs = []
        for _ in range(n_perm):
            p = rng.permutation(n); m = np.zeros(n, bool); m[p[:na]] = True
            diffs.append(np.average(allv[m], weights=w[m]) - np.average(allv[~m], weights=w[~m]))
        diffs = np.array(diffs)
    return float(obs), float(np.mean(np.abs(diffs) >= abs(obs) - 1e-12))
