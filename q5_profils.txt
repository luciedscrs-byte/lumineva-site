def wilson(k, n_, z=1.96):
    if n_ == 0: return [None, None]
    p = k / n_; d = 1 + z * z / n_
    c = (p + z * z / (2 * n_)) / d; h = z * math.sqrt(p * (1 - p) / n_ + z * z / (4 * n_ * n_)) / d
    return [100 * (c - h), 100 * (c + h)]


def fisher(a, na, b, nb):
    return float(stats.fisher_exact([[a, na - a], [b, nb - b]])[1])

def hazard(statut, T, H=30):
    """Observations = (client, commande) où le client a déjà passé `statut` commandes. Silence de T jours depuis cette commande ;
    succès = nouvelle commande dans les H jours suivants. On garde les observations suivies au moins T+H jours."""
    ok = tot = 0
    for v in clients.values():
        for i in range(len(v)):
            if not statut(i + 1): continue
            last = v[i]['dt']; nxt = v[i + 1]['dt'] if i + 1 < len(v) else None
            if (REF - last).days < T + H: continue
            if nxt is not None and (nxt - last).days <= T: continue
            tot += 1; ok += 1 if (nxt is not None and (nxt - last).days <= T + H) else 0
    return {'T': T, 'retours': ok, 'observations': tot, 'taux': 100 * ok / tot if tot else None, 'ic': wilson(ok, tot)}

def base90(filtre_rang, jours=90):
    """Après une commande de rang filtré, combien d'articles et d'euros le client achète-t-il dans les `jours` jours suivants ? (clients suivis au moins `jours` jours)"""
    art, eur_, cmdn, n_ = [], [], [], 0
    for v in clients.values():
        for i in range(len(v)):
            if not filtre_rang(i + 1): continue
            if (REF - v[i]['dt']).days < jours: continue
            sui = [o for o in v[i + 1:] if (o['dt'] - v[i]['dt']).days <= jours]
            art.append(sum(o['qte'] for o in sui)); eur_.append(sum(o['sous_total'] for o in sui)); cmdn.append(len(sui))
    return {'observations': len(art), 'articles_moy': float(np.mean(art)), 'euros_moy': float(np.mean(eur_)), 'commandes_moy': float(np.mean(cmdn))}

# =================================================================== J. MODÈLE DE L'ABONNEMENT CIBLÉ (3 bougies en 3 mois, payé d'avance)
# Pour un client qui achèterait b articles en 90 jours sans l'offre : (3 - b) bougies sont des ventes en plus (marge à p euros),
# min(b, 3) sont des bougies qu'il aurait achetées de toute façon (on perd la remise PL - p) ; on retire les frais de paiement.
MARGES = {'bas': 0.357, 'haut': 0.5}      # 0,50 x part des bougies dans le chiffre d'affaires (71,4 %) ; 0,50 sur tout (voir site/scripts/decisions_charts.py)
PL_ = out['prix_bougie']['mediane_gourmande']
def _fee(p, prepaye=True): return (c0 + c1 * 3 * p) if prepaye else 3 * (c0 + c1 * p)
def _gain(m, p, b, prepaye=True):
    c = PL_ * (1 - m); return max(0, 3 - b) * (p - c) - min(b, 3) * (PL_ - p) - _fee(p, prepaye)
def _seuil(m, p, prepaye=True):
    c = PL_ * (1 - m); return (3 * (p - c) - _fee(p, prepaye)) / (PL_ - c)
bk = {'1': out['base90']['1']['articles_moy'], '2': out['base90']['2']['articles_moy'], '3+': out['base90']['3+']['articles_moy']}
out['abonnement'] = {'marges': MARGES, 'prix_liste': PL_,
                     'gain': {'%d|%s|%s' % (p, k, mk): float(_gain(m, p, b)) for p in (18, 20, 22) for k, b in bk.items() for mk, m in MARGES.items()},
                     'seuil': {'%d|%s' % (p, mk): float(_seuil(m, p)) for p in (18, 20, 22) for mk, m in MARGES.items()},
                     'frais_mensuel_20': float(_fee(20, False)), 'frais_avance_20': float(_fee(20, True))}

def _jo(a, b):
    d = 0; cur = a.date()
    while cur < b.date():
        cur += timedelta(days=1)
        if cur.weekday() < 5: d += 1
    return d
